import hashlib,json,pathlib,subprocess
out=pathlib.Path('/results')
harness=pathlib.Path('/work/packages/apack-ears/bench/sqlite/admission-native-control.ts')
(out/'native-control-provenance.json').write_text(json.dumps({'candidate':'6c29f083f344a2ae4b373c82d0946011e1dbb3a5','elmdb':'6ab2f69fa13289ed4f02952253c58b4d41922c98','harnessSha256':hashlib.sha256(harness.read_bytes()).hexdigest(),'diagnosticOnly':True},indent=2)+'\n')
for trial in range(1,4):
 name='native-control-'+str(trial)
 print(name,flush=True)
 result=subprocess.run(['node','--import','tsx',str(harness),'10000','--durable'],cwd='/work',stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,timeout=300)
 (out/(name+'.stdout')).write_text(result.stdout)
 (out/(name+'.stderr')).write_text(result.stderr)
 assert result.returncode==0,(name,result.returncode,result.stderr)
 value=json.loads(result.stdout)
 (out/(name+'.json')).write_text(json.dumps(value,indent=2)+'\n')
 print(json.dumps({k:value[k] for k in ('backend','operationMs','lookupMs','timerStallMs')}),flush=True)
probe="""import Database from 'better-sqlite3'; import {createHash} from 'node:crypto'; import {readFileSync} from 'node:fs'; const db=new Database(':memory:'); const version=db.prepare('select sqlite_version() v').get(); const bindings=process.report.getReport().sharedObjects.filter(name=>name.endsWith('.node')).map(name=>({name,sha256:createHash('sha256').update(readFileSync(name)).digest('hex')})); console.log(JSON.stringify({version,bindings})); db.close();"""
probe_result=subprocess.run(['node','--input-type=module','-e',probe],cwd='/work',capture_output=True,text=True,check=True)
value=json.loads(probe_result.stdout)
assert len(value['bindings'])>0,value
(out/'sqlite-binding-supplement.json').write_text(json.dumps(value,indent=2)+'\n')
