paths={n:r/('packages/apack-ears/src/sqlite/'+n+'.ts') for n in ['database','search']};original={n:p.read_text() for n,p in paths.items()}
mutations=[
 ('readonly','database','this.search.validateStructures(this.heldSchema);','','trigram table'),
 ('rebuild','database','this.repairSearch(this.search.ensure(this.heldSchema), this.heldSchema);','this.search.ensure(this.heldSchema);','trigram table'),
 ('stale','search','for (const group of missing) if (group.reset) this.db.sql.exec(group.reset);','','stale external-content'),
 ('hash','database',"this.run('DELETE FROM text_meta WHERE entity_id IN (SELECT id FROM entities WHERE type=?)', [type]);",'','text table'),

 ('healthy-open','database',"this.readOnly || !this.search.needsRepair(this.heldSchema)",'this.readOnly','opens healthy storage'),
]
try:

  if before is None:
   a=s.index("        this.sql.exec('BEGIN IMMEDIATE');");b=s.index('      this.indexBuilds',a)

  else:
   assert before in s,name;s=s.replace(before,after)
  paths[file].write_text(s)
  p=run('negative-'+name,['npm','test','-w','@apackdev/ears','--','tests/sqlite/search-recovery.spec.ts','-t',pattern]);assert p.returncode!=0 and 'FAIL ' in p.stdout,(name,p.stdout)
  paths[file].write_text(original[file])
