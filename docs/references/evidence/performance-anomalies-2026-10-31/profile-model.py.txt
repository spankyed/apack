let phases: Record<string, { calls: number; ms: number }> = {};
let profileStart = performance.now();
let profileCpu = process.cpuUsage();
function measured<T>(name: string, fn: () => T): T {
  const start = performance.now();
  try { return fn(); } finally {
    const phase = phases[name] ??= { calls: 0, ms: 0 };
    phase.calls++; phase.ms += performance.now() - start;
  }
}
async function measuredAsync<T>(name: string, fn: () => Promise<T>): Promise<T> {
  const start = performance.now();
  try { return await fn(); } finally {
    const phase = phases[name] ??= { calls: 0, ms: 0 };
    phase.calls++; phase.ms += performance.now() - start;
  }
}
function observed<T extends object>(port: T, prefix: string): T {
  return new Proxy(port, { get(target, key) {
    const value = Reflect.get(target, key, target);
    return typeof value !== 'function' ? value : (...args: unknown[]) => measured(
      `${prefix}.${String(key)}${key === 'commit' || key === 'write' ? ':' + String(args.at(-1)) .slice(0, 8) : ''}`,
      () => { const answer = Reflect.apply(value, target, args); return key === 'begin' ? observed(answer, 'unit') : answer; });
  } });
}
