import { spawn } from 'node:child_process';
import { once } from 'node:events';
import { expect, it, vi } from 'vitest';
import { ProcessManager } from '../packages/main/src/modules/api-server/process-manager.ts';
vi.mock('electron', () => ({ app: { isPackaged: false }, BrowserWindow: { getAllWindows: () => [] } }));
vi.mock('../packages/main/src/modules/api-server/logger.ts', () => ({ logInfo: vi.fn(), logError: vi.fn() }));
it('observes child exit after processing an overdue escalation timer', async () => {
  const child = spawn(process.execPath, ['-e', "process.on('SIGTERM', () => {}); setInterval(() => {}, 1000); process.send('ready');"], { stdio: ['ignore', 'pipe', 'pipe', 'ipc'] });
  try {
    await once(child, 'message');
    const manager = new ProcessManager({});
    manager.setProcess(child);
    const stopping = manager.stop('SIGTERM', 5);
    Atomics.wait(new Int32Array(new SharedArrayBuffer(4)), 0, 0, 1100);
    await expect(stopping).resolves.toBeUndefined();
    expect(manager.isRunning()).toBe(false);
  } finally {
    if (child.exitCode === null && child.signalCode === null) {
      const exited = once(child, 'exit');
      child.kill('SIGKILL');
      await exited;
    }
  }
});
